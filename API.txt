# API Endpoints Documentation

## Authentication APIs

# Đăng ký tài khoản mới
POST /api/auth/register/
Content-Type: application/json
{
    "username": "testuser",
    "email": "test@example.com",
    "password": "Test@123",
    "first_name": "Test",
    "last_name": "User"
}

# Đăng nhập
POST /api/auth/login/
Content-Type: application/json
{
    "username": "testuser",
    "password": "Test@123"
}

# Refresh token
POST /api/auth/refresh/
Content-Type: application/json
{
    "refresh": "your_refresh_token"
}

## Product Category APIs

# Lấy danh sách danh mục
GET /api/categories/
Authorization: Bearer your_access_token

# Tạo danh mục mới
POST /api/categories/
Content-Type: application/json
{
    "name": "Đồng hồ nam",
    "description": "Các loại đồng hồ dành cho nam",
    "parent_id": null,
    "display_order": 1
}

# Lấy chi tiết danh mục
GET /api/categories/{id}/

# Cập nhật danh mục
PUT /api/categories/{id}/
Content-Type: application/json
{
    "name": "Đồng hồ nam cao cấp",
    "description": "Các loại đồng hồ nam cao cấp",
    "display_order": 2
}

# Xóa danh mục
DELETE /api/categories/{id}/

# Lọc danh mục theo parent_id
GET /api/categories/?parent_id=1

# Tìm kiếm danh mục
GET /api/categories/?search=rolex

# Sắp xếp danh mục
GET /api/categories/?ordering=name
GET /api/categories/?ordering=-display_order

# Lấy danh sách danh mục con
GET /api/categories/{id}/subcategories/

## Order APIs

# Lấy danh sách đơn hàng
GET /api/orders/
Authorization: Bearer your_access_token

# Tạo đơn hàng mới
POST /api/orders/
Content-Type: application/json
{
    "store_id": 1,
    "customer_id": 1,
    "order_details": [
        {
            "product_id": 1,
            "quantity": 1,
            "unit_price": 50000000
        }
    ],
    "shipping_address": "Hà Nội",
    "shipping_fee": 50000,
    "payment_method": "cash"
}

# Lấy chi tiết đơn hàng
GET /api/orders/{id}/

# Cập nhật trạng thái đơn hàng
PUT /api/orders/{id}/
Content-Type: application/json
{
    "status": "processing"
}

# Xóa đơn hàng
DELETE /api/orders/{id}/

# Lọc đơn hàng
GET /api/orders/?status=pending&store_id=1&customer_id=2
GET /api/orders/?start_date=2024-03-01&end_date=2024-03-20

# Tìm kiếm đơn hàng
GET /api/orders/?search=0123456789

# Sắp xếp đơn hàng
GET /api/orders/?ordering=-order_date
GET /api/orders/?ordering=-total_amount

# Hủy đơn hàng
POST /api/orders/{id}/cancel/
