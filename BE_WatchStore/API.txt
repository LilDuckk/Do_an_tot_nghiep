## Product Category APIs

# Lấy danh sách danh mục

# Tạo danh mục mới
POST {{base_url}}/categories/
{
    "name": "Đồng hồ nam",
    "description": "Các loại đồng hồ dành cho nam",
    "parent_id": null,
    "display_order": 1
}

GET {{base_url}}/categories/{id}/

# Cập nhật danh mục
PUT {{base_url}}/categories/{id}/
{
    "name": "Đồng hồ nam cao cấp",
    "description": "Các loại đồng hồ nam cao cấp",
    "display_order": 2
}

DELETE {{base_url}}/categories/{id}/

GET {{base_url}}/categories/?parent_id=1

# Tìm kiếm danh mục
GET {{base_url}}/categories/?search=rolex

# Sắp xếp danh mục
GET {{base_url}}/categories/?ordering=name
GET {{base_url}}/categories/?ordering=-display_order
