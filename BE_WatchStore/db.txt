-- WATCH STORE DATABASE SCHEMA

-- Quản lý vai trò và quyền hạn
CREATE TABLE Role (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Permission (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    code VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE RolePermission (
    id SERIAL PRIMARY KEY,
    role_id INTEGER REFERENCES Role(id),
    permission_id INTEGER REFERENCES Permission(id),
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(role_id, permission_id)
);

-- Quản lý người dùng và nhân viên
CREATE TABLE UserAccount (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE,
    role_id INTEGER REFERENCES Role(id),
    last_login TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    failed_login_attempts INTEGER DEFAULT 0,
    account_locked_until TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Thêm bảng quản lý phiên làm việc
CREATE TABLE UserSession (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES UserAccount(id),
    session_token VARCHAR(255) NOT NULL,
    ip_address VARCHAR(50),
    user_agent TEXT,
    login_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expiry_time TIMESTAMP NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Thêm bảng xác thực hai lớp
CREATE TABLE TwoFactorAuth (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES UserAccount(id),
    secret_key VARCHAR(255) NOT NULL,
    is_enabled BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Employee (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES UserAccount(id),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    address TEXT,
    employee_code VARCHAR(50) UNIQUE,
    position VARCHAR(100),
    hire_date DATE,
    store_id INTEGER REFERENCES Store(id), -- Thêm liên kết với cửa hàng
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Quản lý cửa hàng
CREATE TABLE Store (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    address TEXT NOT NULL,
    phone VARCHAR(20),
    store_code VARCHAR(50) UNIQUE NOT NULL,
    manager_id INTEGER REFERENCES Employee(id),
    opening_date DATE,
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng phân quyền theo cửa hàng
CREATE TABLE StorePermission (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES UserAccount(id),
    store_id INTEGER REFERENCES Store(id),
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id),
    UNIQUE(user_id, store_id)
);

-- Quản lý khách hàng
CREATE TABLE Customer (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255),
    phone VARCHAR(20),
    address TEXT,
    birth_date DATE,
    gender VARCHAR(10),
    notes TEXT,
    user_id INTEGER REFERENCES UserAccount(id), -- Thêm liên kết với tài khoản đăng nhập
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng sản phẩm yêu thích của khách hàng
CREATE TABLE Wishlist (
    id SERIAL PRIMARY KEY,
    customer_id INTEGER REFERENCES Customer(id),
    product_id INTEGER REFERENCES Product(id),
    added_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(customer_id, product_id)
);

-- Quản lý danh mục và sản phẩm
CREATE TABLE Category (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    parent_id INTEGER REFERENCES Category(id),
    display_order INTEGER DEFAULT 0,
    slug VARCHAR(255) UNIQUE, -- Thêm slug cho SEO
    meta_title VARCHAR(255), -- Thêm meta cho SEO
    meta_description TEXT, -- Thêm meta cho SEO
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE Brand (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    logo_url VARCHAR(255),
    display_order INTEGER DEFAULT 0,
    slug VARCHAR(255) UNIQUE, -- Thêm slug cho SEO
    meta_title VARCHAR(255), -- Thêm meta cho SEO
    meta_description TEXT, -- Thêm meta cho SEO
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE Product (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    category_id INTEGER REFERENCES Category(id),
    brand_id INTEGER REFERENCES Brand(id),
    base_price DECIMAL(10,2) NOT NULL,
    warranty_period INTEGER, -- số tháng bảo hành
    slug VARCHAR(255) UNIQUE, -- Thêm slug cho SEO
    meta_title VARCHAR(255), -- Thêm meta cho SEO
    meta_description TEXT, -- Thêm meta cho SEO
    is_featured BOOLEAN DEFAULT FALSE, -- Thêm đánh dấu sản phẩm nổi bật
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng thông số kỹ thuật đồng hồ
CREATE TABLE WatchSpecification (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES Product(id),
    case_material VARCHAR(100), -- Chất liệu vỏ
    case_diameter VARCHAR(50), -- Đường kính vỏ
    case_thickness VARCHAR(50), -- Độ dày vỏ
    strap_material VARCHAR(100), -- Chất liệu dây đeo
    movement_type VARCHAR(100), -- Loại máy (Automatic, Quartz...)
    water_resistance VARCHAR(50), -- Độ chống nước
    crystal_type VARCHAR(100), -- Loại mặt kính
    dial_color VARCHAR(50), -- Màu mặt số
    gender VARCHAR(20), -- Giới tính (Men, Women, Unisex)
    functions TEXT, -- Chức năng (lịch, chronograph...)
    origin VARCHAR(100), -- Xuất xứ
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Bảng thuộc tính sản phẩm
CREATE TABLE AttributeType (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE AttributeValue (
    id SERIAL PRIMARY KEY,
    attribute_type_id INTEGER REFERENCES AttributeType(id),
    value VARCHAR(255) NOT NULL,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ProductVariant (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES Product(id),
    sku VARCHAR(100) UNIQUE NOT NULL,
    price_adjustment DECIMAL(10,2) DEFAULT 0, -- điều chỉnh giá so với base_price
    stock_alert_threshold INTEGER DEFAULT 5,
    barcode VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE ProductVariantAttribute (
    id SERIAL PRIMARY KEY,
    product_variant_id INTEGER REFERENCES ProductVariant(id),
    attribute_value_id INTEGER REFERENCES AttributeValue(id),
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(product_variant_id, attribute_value_id)
);

-- Thêm bảng hình ảnh sản phẩm
CREATE TABLE ProductImage (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES Product(id),
    product_variant_id INTEGER REFERENCES ProductVariant(id),
    image_url VARCHAR(255) NOT NULL,
    is_primary BOOLEAN DEFAULT FALSE,
    alt_text VARCHAR(255), -- Văn bản thay thế cho SEO
    display_order INTEGER DEFAULT 0,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng đánh giá sản phẩm
CREATE TABLE ProductReview (
    id SERIAL PRIMARY KEY,
    product_id INTEGER REFERENCES Product(id),
    customer_id INTEGER REFERENCES Customer(id),
    order_id INTEGER REFERENCES Orders(id),
    rating INTEGER CHECK (rating BETWEEN 1 AND 5),
    title VARCHAR(255),
    comment TEXT,
    is_verified_purchase BOOLEAN DEFAULT FALSE,
    is_approved BOOLEAN DEFAULT FALSE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    approved_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng giỏ hàng
CREATE TABLE Cart (
    id SERIAL PRIMARY KEY,
    customer_id INTEGER REFERENCES Customer(id),
    session_id VARCHAR(255), -- Cho khách vãng lai
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE CartItem (
    id SERIAL PRIMARY KEY,
    cart_id INTEGER REFERENCES Cart(id),
    product_variant_id INTEGER REFERENCES ProductVariant(id),
    quantity INTEGER NOT NULL DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Quản lý kho
CREATE TABLE Inventory (
    id SERIAL PRIMARY KEY,
    product_variant_id INTEGER REFERENCES ProductVariant(id),
    store_id INTEGER REFERENCES Store(id),
    quantity INTEGER DEFAULT 0,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(product_variant_id, store_id)
);

CREATE TABLE InventoryTransaction (
    id SERIAL PRIMARY KEY,
    inventory_id INTEGER REFERENCES Inventory(id),
    transaction_type VARCHAR(20) NOT NULL CHECK (transaction_type IN ('STOCK_IN', 'STOCK_OUT', 'ADJUSTMENT')),
    quantity INTEGER NOT NULL,
    unit_price DECIMAL(10,2), -- giá nhập hoặc giá xuất
    reference_id INTEGER, -- ID của đơn hàng hoặc phiếu nhập
    reference_type VARCHAR(50), -- 'ORDER', 'PURCHASE', 'ADJUSTMENT'
    note TEXT,
    transaction_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

-- Thêm bảng kiểm kê kho
CREATE TABLE StockTake (

    start_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

);

CREATE TABLE StockTakeDetail (

-- Thêm bảng chuyển kho giữa các cửa hàng

CREATE TABLE StockTransferDetail (

    tax_code VARCHAR(50),

CREATE TABLE PurchaseOrder (

    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    is_deleted BOOLEAN DEFAULT FALSE,

    unit_price DECIMAL(10,2) NOT NULL,

-- Quản lý đơn hàng
CREATE TABLE Orders (
    id SERIAL PRIMARY KEY,
    customer_id INTEGER REFERENCES Customer(id),
    store_id INTEGER REFERENCES Store(id),
    employee_id INTEGER REFERENCES Employee(id), -- nhân viên bán hàng
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'PENDING' CHECK (status IN ('PENDING', 'CONFIRMED', 'PROCESSING', 'SHIPPED', 'DELIVERED', 'COMPLETED', 'CANCELLED')),
    payment_method VARCHAR(50),
    payment_status VARCHAR(50) DEFAULT 'PENDING' CHECK (payment_status IN ('PENDING', 'PAID', 'FAILED', 'REFUNDED')),
    shipping_address TEXT,
    shipping_method VARCHAR(100),
    tracking_number VARCHAR(100),
    subtotal DECIMAL(10,2) NOT NULL,
    tax DECIMAL(10,2) DEFAULT 0,
    shipping_fee DECIMAL(10,2) DEFAULT 0,
    discount DECIMAL(10,2) DEFAULT 0,
    total_amount DECIMAL(10,2) NOT NULL,
    note TEXT,
    is_online_order BOOLEAN DEFAULT FALSE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE OrderDetail (
    id SERIAL PRIMARY KEY,
    order_id INTEGER REFERENCES Orders(id),
    product_variant_id INTEGER REFERENCES ProductVariant(id),
    quantity INTEGER NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL,
    discount DECIMAL(10,2) DEFAULT 0,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Thêm bảng quản lý mã giảm giá
CREATE TABLE Coupon (

    starts_at TIMESTAMP NOT NULL,

    usage_count INTEGER DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE OrderCoupon (
    id SERIAL PRIMARY KEY,
    order_id INTEGER REFERENCES Orders(id),
    coupon_id INTEGER REFERENCES Coupon(id),

-- Thêm bảng quản lý trả hàng
CREATE TABLE ReturnOrder (

    refund_status VARCHAR(50) DEFAULT 'PENDING' CHECK (refund_status IN ('PENDING', 'COMPLETED', 'FAILED')),
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INTEGER REFERENCES UserAccount(id),
    updated_by INTEGER REFERENCES UserAccount(id)
);

CREATE TABLE ReturnOrderDetail (

    SUM(total_amount) as total_revenue
FROM Orders
WHERE is_deleted = FALSE AND status = 'COMPLETED'

-- View cho sản phẩm bán chạy nhất
CREATE VIEW BestSellingProducts AS
SELECT 
    p.id,
    p.name,
    p.base_price,
    SUM(od.quantity) as total_sold,
    SUM(od.unit_price * od.quantity) as total_revenue
FROM OrderDetail od
JOIN ProductVariant pv ON od.product_variant_id = pv.id
JOIN Product p ON pv.product_id = p.id
JOIN Orders o ON od.order_id = o.id
WHERE o.status = 'COMPLETED' AND o.is_deleted = FALSE
GROUP BY p.id, p.name, p.base_price
ORDER BY total_sold DESC;

    c.last_name,
    c.email,

    SUM(o.total_amount) as total_spent
FROM Customer c

RETURNS TRIGGER AS $$
BEGIN

    VALUES (
        current_setting('app.current_user_id', true)::integer,
        TG_OP,
        TG_TABLE_NAME,
        CASE TG_OP
            WHEN 'DELETE' THEN OLD.id
            ELSE NEW.id
        END,
        CASE WHEN TG_OP = 'DELETE' OR TG
CASE WHEN TG_OP = 'DELETE' OR TG_OP = 'UPDATE' 
            THEN to_jsonb(OLD) 
            ELSE NULL 
        END,

            ELSE NULL 
        END
    );
    RETURN NULL;

-- Thêm trigger để tự động giảm số lượng tồn kho khi đơn hàng hoàn tất
CREATE OR REPLACE FUNCTION reduce_inventory_on_order_complete()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'COMPLETED' AND OLD.status <> 'COMPLETED' THEN
        -- Giảm số lượng tồn kho cho mỗi sản phẩm trong đơn hàng
        INSERT INTO InventoryTransaction (
            inventory_id, 
            transaction_type, 
            quantity, 
            reference_id, 
            reference_type, 
            note,
            created_by
        )
        SELECT 
            i.id,
            'STOCK_OUT',
            od.quantity,
            NEW.id,
            'ORDER',
            'Order completed: ' || NEW.id,
            NEW.updated_by
        FROM OrderDetail od
        JOIN Inventory i ON od.product_variant_id = i.product_variant_id AND i.store_id = NEW.store_id
        WHERE od.order_id = NEW.id;
        
        -- Cập nhật số lượng trong bảng Inventory
        UPDATE Inventory i
        SET quantity = i.quantity - od.quantity,
            last_updated = CURRENT_TIMESTAMP
        FROM OrderDetail od
        WHERE od.order_id = NEW.id
        AND i.product_variant_id = od.product_variant_id
        AND i.store_id = NEW.store_id;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER order_completed_trigger
AFTER UPDATE ON Orders
FOR EACH ROW
EXECUTE FUNCTION reduce_inventory_on_order_complete();

-- Thêm trigger để tăng số lượng tồn kho khi phiếu nhập hoàn tất
CREATE OR REPLACE FUNCTION increase_inventory_on_purchase_delivery()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'DELIVERED' AND OLD.status <> 'DELIVERED' THEN
        -- Tăng số lượng tồn kho cho mỗi sản phẩm trong đơn nhập
        INSERT INTO InventoryTransaction (
            inventory_id, 
            transaction_type, 
            quantity, 
            unit_price,
            reference_id, 
            reference_type, 
            note,
            created_by
        )
        SELECT 
            i.id,
            'STOCK_IN',
            pod.received_quantity,
            pod.unit_price,
            NEW.id,
            'PURCHASE',
            'Purchase delivered: ' || NEW.id,
            NEW.updated_by
        FROM PurchaseOrderDetail pod
        JOIN Inventory i ON pod.product_variant_id = i.product_variant_id AND i.store_id = NEW.store_id
        WHERE pod.purchase_order_id = NEW.id AND pod.received_quantity > 0;
        
        -- Cập nhật số lượng trong bảng Inventory
        UPDATE Inventory i
        SET quantity = i.quantity + pod.received_quantity,
            last_updated = CURRENT_TIMESTAMP
        FROM PurchaseOrderDetail pod
        WHERE pod.purchase_order_id = NEW.id
        AND i.product_variant_id = pod.product_variant_id
        AND i.store_id = NEW.store_id
        AND pod.received_quantity > 0;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER purchase_delivered_trigger
AFTER UPDATE ON PurchaseOrder
FOR EACH ROW
EXECUTE FUNCTION increase_inventory_on_purchase_delivery();

-- Thêm trigger cho việc chuyển kho
CREATE OR REPLACE FUNCTION process_stock_transfer()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'COMPLETED' AND OLD.status <> 'COMPLETED' THEN
        -- Giảm số lượng tại cửa hàng nguồn
        INSERT INTO InventoryTransaction (
            inventory_id, 
            transaction_type, 
            quantity, 
            reference_id, 
            reference_type, 
            note,
            created_by
        )
        SELECT 
            i.id,
            'STOCK_OUT',
            std.quantity,
            NEW.id,
            'TRANSFER_OUT',
            'Stock transfer to store: ' || NEW.destination_store_id,
            NEW.updated_by
        FROM StockTransferDetail std
        JOIN Inventory i ON std.product_variant_id = i.product_variant_id AND i.store_id = NEW.source_store_id
        WHERE std.stock_transfer_id = NEW.id;
        
        UPDATE Inventory i
        SET quantity = i.quantity - std.quantity,
            last_updated = CURRENT_TIMESTAMP
        FROM StockTransferDetail std
        WHERE std.stock_transfer_id = NEW.id
        AND i.product_variant_id = std.product_variant_id
        AND i.store_id = NEW.source_store_id;
        
        -- Tăng số lượng tại cửa hàng đích
        INSERT INTO InventoryTransaction (
            inventory_id, 
            transaction_type, 
            quantity, 
            reference_id, 
            reference_type, 
            note,
            created_by
        )
        SELECT 
            i.id,
            'STOCK_IN',
            std.quantity,
            NEW.id,
            'TRANSFER_IN',
            'Stock transfer from store: ' || NEW.source_store_id,
            NEW.updated_by
        FROM StockTransferDetail std
        JOIN Inventory i ON std.product_variant_id = i.product_variant_id AND i.store_id = NEW.destination_store_id
        WHERE std.stock_transfer_id = NEW.id;
        
        UPDATE Inventory i
        SET quantity = i.quantity + std.quantity,
            last_updated = CURRENT_TIMESTAMP
        FROM StockTransferDetail std
        WHERE std.stock_transfer_id = NEW.id
        AND i.product_variant_id = std.product_variant_id
        AND i.store_id = NEW.destination_store_id;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Procedure để tính tồn kho theo thời gian
CREATE OR REPLACE FUNCTION get_inventory_history(
    p_product_variant_id INTEGER, 
    p_store_id INTEGER,
    p_start_date TIMESTAMP,
    p_end_date TIMESTAMP
)
RETURNS TABLE (
    transaction_date TIMESTAMP,
    quantity_change INTEGER,
    running_balance INTEGER
) AS $$
DECLARE
    v_initial_stock INTEGER;
BEGIN
    -- Lấy số lượng tồn kho trước ngày bắt đầu
    SELECT COALESCE(SUM(CASE 
        WHEN transaction_type = 'STOCK_IN' THEN quantity
        WHEN transaction_type = 'STOCK_OUT' THEN -quantity
        WHEN transaction_type = 'ADJUSTMENT' THEN quantity
    END), 0) INTO v_initial_stock
    FROM InventoryTransaction it
    JOIN Inventory i ON it.inventory_id = i.id
    WHERE i.product_variant_id = p_product_variant_id
      AND i.store_id = p_store_id
      AND it.transaction_date < p_start_date
      AND it.is_deleted = FALSE;

    RETURN QUERY
    WITH transactions AS (
        SELECT 
            it.transaction_date,
            CASE 
                WHEN transaction_type = 'STOCK_IN' THEN quantity
                WHEN transaction_type = 'STOCK_OUT' THEN -quantity
                WHEN transaction_type = 'ADJUSTMENT' THEN quantity
            END as quantity_change
        FROM InventoryTransaction it
        JOIN Inventory i ON it.inventory_id = i.id
        WHERE i.product_variant_id = p_product_variant_id
          AND i.store_id = p_store_id
          AND it.transaction_date BETWEEN p_start_date AND p_end_date
          AND it.is_deleted = FALSE
        ORDER BY it.transaction_date
    )
    SELECT 
        t.transaction_date,
        t.quantity_change,
        v_initial_stock + SUM(t.quantity_change) OVER (ORDER BY t.transaction_date) as running_balance
    FROM transactions t;
END;
$$ LANGUAGE plpgsql;

-- Thêm procedure tính doanh số bán hàng theo nhân viên
CREATE OR REPLACE FUNCTION get_employee_sales(
    p_employee_id INTEGER,
    p_start_date TIMESTAMP,
    p_end_date TIMESTAMP
)
RETURNS TABLE (
    sale_date DATE,
    total_orders INTEGER,
    total_sales DECIMAL(10,2)
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        DATE(o.order_date) as sale_date,
        COUNT(o.id) as total_orders,
        SUM(o.total_amount) as total_sales
    FROM Orders o
    WHERE o.employee_id = p_employee_id
      AND o.status = 'COMPLETED'
      AND o.is_deleted = FALSE
      AND o.order_date BETWEEN p_start_date AND p_end_date
    GROUP BY DATE(o.order_date)
    ORDER BY DATE(o.order_date);
END;
$$ LANGUAGE plpgsql;

-- Thêm procedure báo cáo sản phẩm dưới ngưỡng tồn kho
CREATE OR REPLACE FUNCTION get_low_stock_products(
    p_store_id INTEGER DEFAULT NULL
)
RETURNS TABLE (
    store_name VARCHAR(255),
    product_name VARCHAR(255),
    variant_sku VARCHAR(100),
    current_quantity INTEGER,
    alert_threshold INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        s.name as store_name,
        p.name as product_name,
        pv.sku as variant_sku,
        i.quantity as current_quantity,
        pv.stock_alert_threshold as alert_threshold
    FROM Inventory i
    JOIN Store s ON i.store_id = s.id
    JOIN ProductVariant pv ON i.product_variant_id = pv.id
    JOIN Product p ON pv.product_id = p.id
    WHERE i.quantity <= pv.stock_alert_threshold
      AND (p_store_id IS NULL OR i.store_id = p_store_id)
      AND s.is_active = TRUE
      AND p.is_active = TRUE
      AND pv.is_active = TRUE
    ORDER BY s.name, p.name, pv.sku;
END;
$$ LANGUAGE plpgsql;

-- Thêm function để tạo slug từ text
CREATE OR REPLACE FUNCTION create_slug(text_to_slug TEXT)
RETURNS TEXT AS $$
DECLARE
    result TEXT;
BEGIN
    -- Chuyển sang chữ thường
    result := LOWER(text_to_slug);
    
    -- Loại bỏ các ký tự đặc biệt, thay thế bằng dấu gạch ngang
    result := REGEXP_REPLACE(result, '[^a-z0-9\s]', '', 'g');
    
    -- Thay thế khoảng trắng bằng dấu gạch ngang
    result := REGEXP_REPLACE(result, '\s+', '-', 'g');
    
    -- Loại bỏ các dấu gạch ngang liên tiếp
    result := REGEXP_REPLACE(result, '-+', '-', 'g');
    
    -- Loại bỏ dấu gạch ngang ở đầu và cuối
    result := TRIM(BOTH '-' FROM result);
    
    RETURN result;
END;
$$ LANGUAGE plpgsql;

-- Thêm triggers để tự động tạo slug khi thêm mới hoặc cập nhật
CREATE OR REPLACE FUNCTION generate_product_slug()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.slug IS NULL OR NEW.slug = '' THEN

    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
